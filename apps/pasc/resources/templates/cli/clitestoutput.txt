/// <summary> Captures the output of a command builder, so the tests can check what a command
/// shows without writing to the console. </summary>
unit TestOutput;

{$MODE DELPHI}{$H+}

interface

uses
  Command.Interfaces;

  /// <summary> Redirects the output of the builder to OutputCaptured, cleaning it. </summary>
  procedure CaptureOutput(ABuilder: ICommandBuilder);

var
  OutputCaptured: string;

implementation

procedure CaptureText(const AMessage: string);
begin
  OutputCaptured := OutputCaptured + AMessage;
end;

procedure CaptureColorText(const AMessage: string; const AColor: Byte);
begin
  OutputCaptured := OutputCaptured + AMessage;
end;

procedure CaptureOutput(ABuilder: ICommandBuilder);
begin
  OutputCaptured := '';
  ABuilder.Output := CaptureText;
  ABuilder.OutputColor := CaptureColorText;
end;

end.
